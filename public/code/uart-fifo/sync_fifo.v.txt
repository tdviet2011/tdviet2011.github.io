// FIFO đồng bộ, kiểu first-word fall-through:
// khi empty = 0 thì rd_data đã là phần tử đầu hàng đợi; rd_en lấy phần tử đó ra.
module sync_fifo #(
    parameter WIDTH = 8,
    parameter DEPTH = 16                       // nên là lũy thừa của 2
) (
    input                          clk,
    input                          rst_n,
    input                          wr_en,
    input      [WIDTH-1:0]         wr_data,
    input                          rd_en,
    output     [WIDTH-1:0]         rd_data,
    output                         full,
    output                         empty,
    output     [$clog2(DEPTH+1)-1:0] count
);
    localparam AW = $clog2(DEPTH);              // số bit địa chỉ
    localparam CW = $clog2(DEPTH + 1);          // đếm được tới DEPTH

    reg [WIDTH-1:0] mem [0:DEPTH-1];
    reg [AW-1:0]    wp, rp;
    reg [CW-1:0]    cnt;

    wire do_wr = wr_en && !full;
    wire do_rd = rd_en && !empty;

    assign full    = (cnt == DEPTH);
    assign empty   = (cnt == 0);
    assign count   = cnt;
    assign rd_data = mem[rp];

    // con trỏ và bộ đếm
    always @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            wp  <= 0;
            rp  <= 0;
            cnt <= 0;
        end else begin
            if (do_wr) wp <= wp + 1'b1;         // tự quay vòng khi DEPTH là lũy thừa của 2
            if (do_rd) rp <= rp + 1'b1;
            if (do_wr && !do_rd)      cnt <= cnt + 1'b1;
            else if (do_rd && !do_wr) cnt <= cnt - 1'b1;
        end
    end

    // bộ nhớ: không reset để công cụ suy ra LUTRAM
    always @(posedge clk)
        if (do_wr) mem[wp] <= wr_data;
endmodule
