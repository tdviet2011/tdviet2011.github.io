module baud_gen#(
    parameter CLK_FREQ = 100_000_000,
    parameter BAUD = 115200                       
)(
input clk,
input rst_n,
output reg tick16
    );
    localparam DIV = CLK_FREQ / (BAUD * 16);
    reg [$clog2(DIV):0] cnt;
    always @( posedge clk or negedge rst_n) begin
    tick16 <= 1'b0;
        if(!rst_n) begin
            tick16 <= 1'b0;
            cnt    <= 1'b0;
        end else begin
                    cnt <= cnt +1'b1;
                    if( cnt == DIV - 1 ) begin
                        cnt    <= 0;
                        tick16 <= 1;
                    end
                 end
    end
endmodule
