`timescale 1ns / 1ps
// ============================================================================
// AES_SECURE_TB : Testbench tu kiem tra cho aes256_enc_dec_top (ban hardened)
//   Phan 1 : vector chuan NIST FIPS-197 / SP800-38A, chay back-to-back
//   Phan 2 : kich ban tan cong & loi giao tiep (phai bao fault, khong ro ri)
// Chay: xsim / iverilog -g2005 -o tb AES_SECURE_TB.v <tat ca file .v> && vvp tb
// ============================================================================
module AES_SECURE_TB;

    reg          clk = 0, reset = 0, start = 0, mode = 0;
    reg  [255:0] key;
    reg  [127:0] data_in;
    wire [127:0] data_out;
    wire         done, busy, fault;

    integer err = 0, n, d, i, nz, runs = 0, r, j, b, t, v;

    // dem pass/total theo tung kich ban
    localparam SA = 0, SC = 1, SD = 2, SE1 = 3, SE2 = 4, SE3 = 5, SB = 6, SF = 7, SH = 8, NSCN = 9;
    integer tot [0:NSCN-1];
    integer ps  [0:NSCN-1];

    aes256_enc_dec_top uut (
        .clk(clk), .reset(reset), .start(start), .mode(mode),
        .key(key), .data_in(data_in),
        .data_out(data_out), .done(done), .busy(busy), .fault(fault)
    );

    always #5 clk = ~clk;                      // 100 MHz
    always @(posedge busy) runs = runs + 1;

    localparam [255:0] K0 = 256'h000102030405060708090A0B0C0D0E0F101112131415161718191A1B1C1D1E1F;
    localparam [127:0] P0 = 128'h00112233445566778899AABBCCDDEEFF;
    localparam [127:0] C0 = 128'h8EA2B7CA516745BFEAFC49904B496089;

    task pulse;    begin @(negedge clk); start = 1; @(negedge clk); start = 0; end endtask
    task waitdone; begin n = 0; while (!done && n < 400) begin @(negedge clk); n = n + 1; end end endtask

    task op(input m, input [255:0] k, input [127:0] dat);
    begin
        @(negedge clk); mode = m; key = k; data_in = dat; start = 1;
        @(negedge clk); start = 0;
        @(negedge clk); while (!done) @(negedge clk);
        repeat (2) @(negedge clk);
    end
    endtask

    task rec(input integer k, input ok);
    begin
        tot[k] = tot[k] + 1;
        if (ok) ps[k] = ps[k] + 1;
        else begin err = err + 1; $display("  FAIL kich ban %0d case %0d", k, tot[k]); end
    end
    endtask

    // sau moi lan bi tiem loi: 1 phep sach phai dung (B), roi trang thai phai sach (F)
    task recover;
    begin
        repeat (3) @(negedge clk);
        key = K0; data_in = P0; mode = 1'b0;
        pulse; waitdone; repeat (2) @(negedge clk);
        rec(SB, !fault && data_out === C0);
        nz = 0;
        for (i = 0; i < 60; i = i + 1) if (uut.u_kexp.w[i] !== 32'd0) nz = nz + 1;
        rec(SF, nz == 0 && uut.u_enc.state_reg === 0 && uut.u_dec.state_reg === 0 && uut.data_r === 0);
        repeat (3) @(negedge clk);
    end
    endtask

    task chk(input [255:0] k, input [127:0] p, input [127:0] c);
    begin
        op(1'b0, k, p);
        if (data_out !== c || fault) begin
            err = err + 1; $display("  ENC FAIL exp=%h got=%h fault=%b", c, data_out, fault);
        end
        op(1'b1, k, c);
        if (data_out !== p || fault) begin
            err = err + 1; $display("  DEC FAIL exp=%h got=%h fault=%b", p, data_out, fault);
        end
    end
    endtask

    initial begin
        #22 reset = 1;

        //====================================================================
        $display("\n--- Phan 1: vector chuan NIST (enc + dec, khong reset giua cac lan) ---");
        //====================================================================
        chk(K0, P0, C0);                                                     // FIPS-197 C.3
        chk(256'h603DEB1015CA71BE2B73AEF0857D77811F352C073B6108D72D9810A30914DFF4,
            128'h6BC1BEE22E409F96E93D7E117393172A, 128'hF3EED1BDB5D2A03C064B5A7E3DB181F8);
        chk(256'h603DEB1015CA71BE2B73AEF0857D77811F352C073B6108D72D9810A30914DFF4,
            128'hAE2D8A571E03AC9C9EB76FAC45AF8E51, 128'h591CCB10D410ED26DC5BA74A31362870);
        chk(256'h603DEB1015CA71BE2B73AEF0857D77811F352C073B6108D72D9810A30914DFF4,
            128'h30C81C46A35CE411E5FBC1191A0A52EF, 128'hB6ED21B99CA6F4F9F153E7B1BEAFED1D);
        chk(256'h603DEB1015CA71BE2B73AEF0857D77811F352C073B6108D72D9810A30914DFF4,
            128'hF69F2445DF4F9B17AD2B417BE66C3710, 128'h23304B7A39F9F3FF067D8D8F9E24ECC7);
        chk(256'd0, 128'd0, 128'hDC95C078A2408989AD48A21492842087);          // all-zero
        $display("NIST: %0d loi / 12 phep", err);

        //====================================================================
        $display("\n--- Phan 2: kich ban tan cong / loi giao tiep (quet nhieu case) ---");
        //====================================================================
        for (i = 0; i < NSCN; i = i + 1) begin tot[i] = 0; ps[i] = 0; end
        key = K0; data_in = P0; mode = 1'b0;

        // A. Doi key/data/mode d chu ky sau start (d = 1..10) -> input da CHOT
        for (d = 1; d <= 10; d = d + 1) begin
            pulse; repeat (d) @(negedge clk); data_in = 0; key = 0; mode = 1'b1;
            waitdone;
            rec(SA, data_out === C0 && !fault);
            data_in = P0; key = K0; mode = 1'b0; repeat (3) @(negedge clk);
        end

        // C. Bom start lan 2 (key sai) khi dang busy, d = 1..45
        for (d = 1; d <= 45; d = d + 1) begin
            pulse;
            repeat (d) @(negedge clk);
            if (busy) begin key = ~K0; pulse; key = K0; end
            waitdone; repeat (3) @(negedge clk);
            rec(SC, data_out === C0 && !fault && !busy);
            repeat (5) @(negedge clk);
        end

        // D. Doi mode giua chung, d = 5,10,..,45 -> ket qua theo mode da chot
        for (d = 5; d <= 45; d = d + 5) begin
            pulse; repeat (d) @(negedge clk); mode = 1'b1; waitdone; mode = 1'b0;
            rec(SD, done && data_out === C0 && !fault);
            repeat (3) @(negedge clk);
        end

        // E1. Bo bot vong: ep FSM nhay ROUNDS -> FINAL o vong r = 1..12
        //     ca core ma hoa (lan tinh chinh) va core giai ma (lan kiem tra nguoc)
        //     r = 13 KHONG phai tan cong: o canh clock do FSM von tu chuyen sang
        //     FINAL, nen force FINAL khong bo vong nao (ket qua dung, khong fault).
        for (r = 1; r <= 12; r = r + 1) begin
            pulse;
            t = 0; while (!(uut.st == 3'd2 && uut.u_enc.state == 3'd2 && uut.u_enc.round_num == r) && t < 500) begin @(negedge clk); t = t + 1; end
            @(posedge clk); #1 force uut.u_enc.state = 3'd3;
            @(posedge clk); #1 release uut.u_enc.state;
            waitdone; rec(SE1, fault && data_out === 128'd0); recover;
            pulse;
            t = 0; while (!(uut.st == 3'd3 && uut.u_dec.state == 3'd2 && uut.u_dec.dec_round_num == r) && t < 500) begin @(negedge clk); t = t + 1; end
            @(posedge clk); #1 force uut.u_dec.state = 3'd3;
            @(posedge clk); #1 release uut.u_dec.state;
            waitdone; rec(SE1, fault && data_out === 128'd0); recover;
        end

        // E2. DFA: lat 1 bit state_reg tai vong r = 1..13, bit b thuoc 5 vi tri
        for (r = 1; r <= 13; r = r + 1) begin
            for (j = 0; j < 5; j = j + 1) begin
                b = (j == 0) ? 0 : (j == 1) ? 31 : (j == 2) ? 64 : (j == 3) ? 100 : 127;
                pulse;
                t = 0; while (!(uut.st == 3'd2 && uut.u_enc.state == 3'd2 && uut.u_enc.round_num == r) && t < 500) begin @(negedge clk); t = t + 1; end
                @(posedge clk); #1 uut.u_enc.state_reg = uut.u_enc.state_reg ^ (128'd1 << b);
                waitdone; rec(SE2, fault && data_out === 128'd0); recover;
            end
        end

        // E3. Trang thai FSM la (5,6,7) trong core ma hoa va core giai ma
        for (v = 5; v <= 7; v = v + 1) begin
            pulse;
            t = 0; while (!(uut.st == 3'd2 && uut.u_enc.state == 3'd2) && t < 500) begin @(negedge clk); t = t + 1; end
            @(posedge clk); #1 force uut.u_enc.state = v;
            @(posedge clk); #1 release uut.u_enc.state;
            waitdone; rec(SE3, fault && data_out === 128'd0); recover;
            pulse;
            t = 0; while (!(uut.st == 3'd3 && uut.u_dec.state == 3'd2) && t < 500) begin @(negedge clk); t = t + 1; end
            @(posedge clk); #1 force uut.u_dec.state = v;
            @(posedge clk); #1 release uut.u_dec.state;
            waitdone; rec(SE3, fault && data_out === 128'd0); recover;
        end

        // H. Giu start = 1 lien tuc L chu ky -> chi duoc chay DUNG 1 lan
        for (j = 0; j < 4; j = j + 1) begin
            key = K0; data_in = P0; mode = 1'b0; runs = 0;
            @(negedge clk) start = 1;
            repeat ((j == 0) ? 60 : (j == 1) ? 150 : (j == 2) ? 300 : 1000) @(negedge clk);
            start = 0; repeat (60) @(negedge clk);
            rec(SH, runs == 1);
        end

        // ---------------- bang tong ket (dinh dang de script doc) ----------------
        $display("\n SCN | ten                                   | pass/total");
        $display("SCN A  doi key/data/mode sau start        %0d/%0d", ps[SA],  tot[SA]);
        $display("SCN C  start lan 2 khi busy               %0d/%0d", ps[SC],  tot[SC]);
        $display("SCN D  doi mode giua chung                %0d/%0d", ps[SD],  tot[SD]);
        $display("SCN E1 bo bot vong (enc + dec)            %0d/%0d", ps[SE1], tot[SE1]);
        $display("SCN E2 lat 1 bit state (DFA)              %0d/%0d", ps[SE2], tot[SE2]);
        $display("SCN E3 trang thai FSM la                  %0d/%0d", ps[SE3], tot[SE3]);
        $display("SCN B  hoat dong lai sau fault            %0d/%0d", ps[SB],  tot[SB]);
        $display("SCN F  zeroize sau moi phep               %0d/%0d", ps[SF],  tot[SF]);
        $display("SCN H  giu start=1 (60..1000 chu ky)      %0d/%0d", ps[SH],  tot[SH]);

        $display("\n=========================================");
        $display(" TONG KET: %0d loi   -> %s", err, err == 0 ? "TAT CA PASS" : "CO LOI");
        $display(" Latency 1 phep (VERIFY_EN=1, co kiem tra key schedule) = 66 chu ky");
        $display("=========================================\n");
        $finish;
    end
endmodule
